1. MÔ TẢ CHỨC NĂNG & MỤC ĐÍCH SỬ DỤNG
--------------------------------------------------------------------------------
- Động cơ xác thực Google OAuth & Đồng bộ đám mây 2 chiều chuyên nghiệp cho ứng dụng di động:
  + Trên Android Native: Sử dụng Google Play Services Native SDK (`@codetrix-studio/capacitor-google-auth`) hiển thị cửa sổ chọn tài khoản của điện thoại.
  + Trên Web / Browser: Sử dụng popup OAuth 2.0 hoặc redirect flow chuẩn Google.
- Cơ chế Giữ phiên dài hạn (Persistent Long-Term Session):
  + Tự động xin quyền `offlineAccess` để lấy `serverAuthCode` và `refreshToken`.
  + Lưu Access Token và Refresh Token vào Secure Storage (Android KeyStore).
  + Cơ chế Silent Refresh: Tự động gia hạn Access Token ngầm khi mở app mà người dùng không phải bấm đăng nhập lại.
  + Chống treo mạng: Áp dụng `Promise.race` timeout (8 giây) để tránh tình trạng màn hình quay tròn vô tận nếu dịch vụ Google Play Service trên máy phản hồi chậm.
- Tương tác với Google Drive REST API & Sheets REST API:
  + Tự động tìm kiếm file lưu trữ của ứng dụng trên Drive (Auto-discover Central Hub).
  + Tự động tạo mới file Google Sheets nếu tài khoản chưa có.
  + Đọc toàn bộ ma trận dữ liệu (Matrix Values) từ Google Sheet về đối tượng JSON.
  + Ghi dữ liệu từ ứng dụng lên Google Sheet theo lô (Batch Update) cực nhanh.

```ts
}
```

```ts
import { Capacitor } from '@capacitor/core';
import { GoogleAuth } from '@codetrix-studio/capacitor-google-auth';
import { setSecureItem, getSecureItem, removeSecureItem } from './secureStorage';

const TOKEN_KEY = 'gauth_access_token';
const EXPIRES_AT_KEY = 'gauth_expires_at';
const USER_KEY = 'gauth_user_profile';

export interface GoogleUserProfile {
  email: string;
  name: string;
  imageUrl?: string;
}

let memoryToken: string | null = null;
let memoryExpiresAt: number = 0;

export const GoogleAuthEngine = {
  /**
   * Khởi tạo plugin GoogleAuth cho Android
   */
  async initialize(webClientId: string) {
    if (Capacitor.isNativePlatform()) {
      try {
        await GoogleAuth.initialize({
          clientId: webClientId,
          serverClientId: webClientId,
          scopes: [
            'email',
            'profile',
            'https://www.googleapis.com/auth/drive.file',
            'https://www.googleapis.com/auth/spreadsheets',
          ],
          grantOfflineAccess: true,
        });
      } catch (err) {
        console.warn('[GoogleAuthEngine] Init warning:', err);
      }
    }
  },

  /**
   * Đăng nhập tương tác bằng tài khoản Google
   */
  async signIn(webClientId: string): Promise<{ token: string; user: GoogleUserProfile }> {
    await this.initialize(webClientId);

    if (Capacitor.isNativePlatform()) {
      // Đặt timeout 3 giây cho signOut để tránh treo native picker
      try {
        const signOutPromise = GoogleAuth.signOut();
        const timeoutPromise = new Promise((resolve) => setTimeout(resolve, 3000));
        await Promise.race([signOutPromise, timeoutPromise]);
      } catch {}

      const nativeUser = await GoogleAuth.signIn();
      const token = nativeUser.authentication?.accessToken || '';
      const expiresAt = Date.now() + 3500 * 1000;

      const profile: GoogleUserProfile = {
        email: nativeUser.email || '',
        name: nativeUser.displayName || nativeUser.name || 'Người dùng Google',
        imageUrl: nativeUser.imageUrl || undefined,
      };

      await this.saveSession(token, expiresAt, profile);
      return { token, user: profile };
    } else {
      // Fallback cho Web: Mở OAuth popup hoặc gọi thư viện Google Identity Services
      throw new Error('Vui lòng chạy trên thiết bị Android hoặc cấu hình Web GIS Client.');
    }
  },

  /**
   * Gia hạn token ngầm (Silent Refresh) khi mở app
   */
  async getValidToken(): Promise<string | null> {
    // Nếu token trong RAM còn hạn trên 5 phút
    if (memoryToken && memoryExpiresAt > Date.now() + 300 * 1000) {
      return memoryToken;
    }

    // Đọc từ KeyStore phần cứng
    const storedToken = await getSecureItem(TOKEN_KEY);
    const storedExpiresAt = Number(await getSecureItem(EXPIRES_AT_KEY)) || 0;

    if (storedToken && storedExpiresAt > Date.now() + 300 * 1000) {
      memoryToken = storedToken;
      memoryExpiresAt = storedExpiresAt;
      return storedToken;
    }

    // Nếu hết hạn, thử refresh tự động với timeout 8 giây
    if (Capacitor.isNativePlatform()) {
      try {
        const refreshPromise = GoogleAuth.refresh();
        const timeoutPromise = new Promise<any>((_, reject) =>
          setTimeout(() => reject(new Error('TIMEOUT_REFRESH')), 8000)
        );
        const res = await Promise.race([refreshPromise, timeoutPromise]);
        if (res && res.accessToken) {
          const expiresAt = Date.now() + 3500 * 1000;
          await setSecureItem(TOKEN_KEY, res.accessToken);
          await setSecureItem(EXPIRES_AT_KEY, String(expiresAt));
          memoryToken = res.accessToken;
          memoryExpiresAt = expiresAt;
          return res.accessToken;
        }
      } catch (err) {
        console.warn('[GoogleAuthEngine] Silent refresh thất bại:', err);
      }
    }

    return null;
  },

  /**
   * Lưu phiên vào Secure Storage
   */
  async saveSession(token: string, expiresAt: number, profile: GoogleUserProfile) {
    memoryToken = token;
    memoryExpiresAt = expiresAt;
    await setSecureItem(TOKEN_KEY, token);
    await setSecureItem(EXPIRES_AT_KEY, String(expiresAt));
    await setSecureItem(USER_KEY, JSON.stringify(profile));
  },

  /**
   * Đăng xuất
   */
  async signOut() {
    memoryToken = null;
    memoryExpiresAt = 0;
    try {
      if (Capacitor.isNativePlatform()) {
        await GoogleAuth.signOut();
      }
    } catch {}
    await removeSecureItem(TOKEN_KEY);
    await removeSecureItem(EXPIRES_AT_KEY);
    await removeSecureItem(USER_KEY);
  },
};
```

File B: src/utils/googleDriveClient.ts (REST API Client)

```ts
export interface DriveFileItem {
  id: string;
  name: string;
  webViewLink?: string;
  createdTime?: string;
  modifiedTime?: string;
}

export class GoogleDriveClient {
  private accessToken: string;

  constructor(accessToken: string) {
    this.accessToken = accessToken;
  }

  private getHeaders() {
    return {
      Authorization: `Bearer ${this.accessToken}`,
      'Content-Type': 'application/json',
    };
  }

  /**
   * Tìm kiếm file bảng tính Google Sheets của ứng dụng trên Drive
   */
  async findSpreadsheetByName(fileName: string): Promise<DriveFileItem | null> {
    const q = encodeURIComponent(
      `name = '${fileName}' and mimeType = 'application/vnd.google-apps.spreadsheet' and trashed = false`
    );
    const url = `https://www.googleapis.com/drive/v3/files?q=${q}&fields=files(id,name,webViewLink,createdTime,modifiedTime)&spaces=drive`;

    const res = await fetch(url, { headers: this.getHeaders() });
    if (!res.ok) throw new Error(`Lỗi tìm file Drive (HTTP ${res.status})`);

    const data = await res.json();
    if (data.files && data.files.length > 0) {
      return data.files[0];
    }
    return null;
  }

  /**
   * Tạo mới một bảng tính Google Sheets trên Drive
   */
  async createSpreadsheet(title: string): Promise<{ id: string; webViewLink: string }> {
    const createUrl = 'https://sheets.googleapis.com/v4/spreadsheets';
    const body = {
      properties: { title },
      sheets: [{ properties: { title: 'Sheet1' } }],
    };

    const res = await fetch(createUrl, {
      method: 'POST',
      headers: this.getHeaders(),
      body: JSON.stringify(body),
    });

    if (!res.ok) throw new Error(`Không thể tạo bảng tính (HTTP ${res.status})`);
    const data = await res.json();

    const fileId = data.spreadsheetId;
    const webViewLink = data.spreadsheetUrl || `https://docs.google.com/spreadsheets/d/${fileId}/edit`;

    return { id: fileId, webViewLink };
  }

  /**
   * Đọc toàn bộ dữ liệu của một Sheet
   */
  async readSheetData(spreadsheetId: string, range: string = 'Sheet1!A1:Z500'): Promise<any[][]> {
    const url = `https://sheets.googleapis.com/v4/spreadsheets/${spreadsheetId}/values/${encodeURIComponent(
      range
    )}?valueRenderOption=UNFORMATTED_VALUE`;

    const res = await fetch(url, { headers: this.getHeaders() });
    if (!res.ok) throw new Error(`Lỗi đọc dữ liệu Google Sheet (HTTP ${res.status})`);

    const data = await res.json();
    return data.values || [];
  }

  /**
   * Ghi toàn bộ dữ liệu ma trận lên Sheet
   */
  async writeSheetData(
    spreadsheetId: string,
    values: any[][],
    range: string = 'Sheet1!A1'
  ): Promise<boolean> {
    const url = `https://sheets.googleapis.com/v4/spreadsheets/${spreadsheetId}/values/${encodeURIComponent(
      range
    )}?valueInputOption=USER_ENTERED`;

    const res = await fetch(url, {
      method: 'PUT',
      headers: this.getHeaders(),
      body: JSON.stringify({
        range,
        majorDimension: 'ROWS',
        values,
      }),
    });

    if (!res.ok) throw new Error(`Lỗi ghi dữ liệu Google Sheet (HTTP ${res.status})`);
    return true;
  }
}
```

### BƯỚC 3: Đồng bộ dữ liệu chỉ trong vài dòng code
```tsx
import React, { useState } from 'react';
import { GoogleAuthEngine } from './utils/googleAuthEngine';
import { GoogleDriveClient } from './utils/googleDriveClient';

const WEB_CLIENT_ID = 'xxxxxxxx-xxxx.apps.googleusercontent.com';

export default function DriveSyncDemo() {
  const [status, setStatus] = useState('');

  const handleSync = async () => {
    setStatus('Đang lấy token đăng nhập...');
    let token = await GoogleAuthEngine.getValidToken();

    // Nếu chưa đăng nhập hoặc token hết hạn, gọi đăng nhập tương tác
    if (!token) {
      const res = await GoogleAuthEngine.signIn(WEB_CLIENT_ID);
      token = res.token;
    }

    const driveClient = new GoogleDriveClient(token);

    setStatus('Đang tìm kiếm file dữ liệu trên Drive...');
    let file = await driveClient.findSpreadsheetByName('Du_Lieu_App_Cua_Toi');

    if (!file) {
      setStatus('Đang tạo file Google Sheet mới...');
      file = await driveClient.createSpreadsheet('Du_Lieu_App_Cua_Toi');
    }

    setStatus('Đang đẩy dữ liệu lên Google Sheets...');
    const dataMatrix = [
      ['ID', 'Tên Mục', 'Giá Trị', 'Ngày Tạo'],
      ['1', 'Mục mẫu 1', 1500000, new Date().toLocaleDateString('vi-VN')],
      ['2', 'Mục mẫu 2', 3200000, new Date().toLocaleDateString('vi-VN')],
    ];

    await driveClient.writeSheetData(file.id, dataMatrix);
    setStatus(`Đồng bộ thành công! File ID: ${file.id}`);
  };

  return (
    <div className="p-4">
      <button onClick={handleSync} className="px-4 py-2 bg-emerald-600 text-white rounded-lg">
        Đồng bộ lên Google Drive / Sheets
      </button>
      <p className="mt-2 text-sm text-slate-600">{status}</p>
    </div>
  );
}
```

1. MÔ TẢ CHỨC NĂNG & MỤC ĐÍCH SỬ DỤNG
--------------------------------------------------------------------------------
- Cung cấp màn hình chào đón và đăng nhập đa nền tảng hoàn chỉnh cho ứng dụng Mobile & Web:
  + Tự động nhận diện môi trường: Nếu chạy trên Android Native (`Capacitor.isNativePlatform()`), kích hoạt Native Google Play Services Account Picker mượt mà.
  + Nếu chạy trên Web Desktop / Mobile Browser: Kích hoạt Web Google OAuth popup hoặc redirect flow.
  + Chế độ Khách Ngoại Tuyến (Offline Guest Mode): Cho phép người dùng trải nghiệm ngay lập tức các tính năng cục bộ mà không cần tài khoản hay kết nối Internet.
  + Tích hợp Thanh trạng thái Phiên bản & Kiểm tra Cập nhật APK: Hiển thị số phiên bản hiện tại, nút "Kiểm tra cập nhật" và huy hiệu nhấp nháy "Có bản v... • Tải ngay" khi có bản mới từ GitHub.
  + Công cụ chẩn đoán SHA-1 & Google Client ID tích hợp sẵn (In-App SHA-1 Helper):
    * Khi gặp lỗi mã 10 (Developer Error) hoặc 12500 trên Android (do chưa cấu hình fingerprint trên Google Cloud / Firebase Console), modal cung cấp sẵn nút bấm mở hộp thoại chẩn đoán.
    * Người dùng / Lập trình viên có thể xem và bấm copy 1-chạm: SHA-1 Fingerprint, Package Name, Google Client ID để dán thẳng vào Firebase mà không cần kết nối máy tính chạy adb/logcat.

```tsx
import React, { useState } from 'react';
import {
  ShieldCheck,
  AlertCircle,
  RefreshCw,
  HardDrive,
  Copy,
  Check,
  X,
  ArrowUpCircle,
  ExternalLink,
  Info,
} from 'lucide-react';
import { Capacitor } from '@capacitor/core';

export interface AuthUserProfile {
  email: string;
  name: string;
  photoURL?: string;
  role: 'admin' | 'editor' | 'viewer';
  isOffline: boolean;
}

export interface LoginModalProps {
  isOpen: boolean;
  appName?: string;
  appDescription?: string;
  appLogo?: React.ReactNode;
  currentVersion?: string;
  hasNewUpdate?: boolean;
  newVersion?: string;
  serverClientId?: string;
  androidPackageName?: string;
  fixedSha1?: string;
  onLogin: (user: AuthUserProfile, accessToken?: string) => void;
  onGoogleSignIn: () => Promise<{ user: { email: string; displayName?: string; photoURL?: string }; accessToken?: string }>;
  onCheckUpdate?: () => void;
  onOpenUpdateModal?: () => void;
}

export const LoginModal: React.FC<LoginModalProps> = ({
  isOpen,
  appName = 'Ứng Dụng Của Tôi',
  appDescription = 'Quản lý thông minh và an toàn trên thiết bị của bạn',
  appLogo,
  currentVersion = '1.0.0',
  hasNewUpdate = false,
  newVersion,
  serverClientId = 'YOUR_WEB_CLIENT_ID.apps.googleusercontent.com',
  androidPackageName = 'com.mycompany.myapp',
  fixedSha1 = 'AA:BB:CC:DD:EE:FF:00:11:22:33:44:55:66:77:88:99:AA:BB:CC:DD',
  onLogin,
  onGoogleSignIn,
  onCheckUpdate,
  onOpenUpdateModal,
}) => {
  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);
  const [showShaHelp, setShowShaHelp] = useState<boolean>(false);
  const [copiedKey, setCopiedKey] = useState<string | null>(null);

  const isNative = Capacitor.isNativePlatform();

  if (!isOpen) return null;

  const handleCopy = (text: string, key: string) => {
    navigator.clipboard.writeText(text);
    setCopiedKey(key);
    setTimeout(() => setCopiedKey(null), 2000);
  };

  const handleSignIn = async () => {
    setIsLoading(true);
    setErrorMessage(null);
    try {
      const { user, accessToken } = await onGoogleSignIn();
      const cleanEmail = (user.email || '').trim().toLowerCase();
      const authUser: AuthUserProfile = {
        email: cleanEmail,
        name: user.displayName || cleanEmail.split('@')[0] || 'Người dùng',
        photoURL: user.photoURL,
        role: 'admin',
        isOffline: false,
      };
      onLogin(authUser, accessToken);
    } catch (err: any) {
      console.warn('Login error:', err);
      setErrorMessage(
        err?.message || 'Đăng nhập Google thất bại hoặc cửa sổ bị đóng. Vui lòng thử lại.'
      );
    } finally {
      setIsLoading(false);
    }
  };

  const handleOfflineMode = () => {
    const offlineUser: AuthUserProfile = {
      email: 'offline@guest.local',
      name: 'Khách Ngoại Tuyến',
      role: 'admin',
      isOffline: true,
    };
    onLogin(offlineUser);
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/80 backdrop-blur-md animate-in fade-in duration-200">
      <div className="w-full max-w-md bg-slate-900 border border-slate-800 rounded-3xl p-6 sm:p-8 shadow-2xl relative overflow-hidden">
        {/* Trang trí nền */}
        <div className="absolute -right-16 -top-16 w-48 h-48 bg-emerald-500/10 rounded-full blur-3xl pointer-events-none" />
        <div className="absolute -left-16 -bottom-16 w-48 h-48 bg-blue-500/10 rounded-full blur-3xl pointer-events-none" />

        {/* Header Thương hiệu */}
        <div className="text-center relative z-10 mb-8">
          <div className="w-20 h-20 mx-auto mb-4 rounded-2xl bg-gradient-to-tr from-emerald-500 to-teal-400 p-0.5 shadow-lg shadow-emerald-500/20">
            <div className="w-full h-full bg-slate-900 rounded-[14px] flex items-center justify-center overflow-hidden">
              {appLogo || <ShieldCheck className="w-10 h-10 text-emerald-400" />}
            </div>
          </div>
          <h2 className="text-2xl font-black text-white tracking-tight">{appName}</h2>
          <p className="text-slate-400 text-xs mt-1.5 leading-relaxed">{appDescription}</p>
        </div>

        {/* Thông báo lỗi nếu có */}
        {errorMessage && (
          <div className="mb-6 p-4 rounded-2xl bg-rose-500/10 border border-rose-500/20 text-rose-300 text-xs flex items-start gap-3">
            <AlertCircle className="w-5 h-5 shrink-0 text-rose-400 mt-0.5" />
            <div className="flex-1">
              <p className="font-semibold">{errorMessage}</p>
              {isNative && (
                <button
                  type="button"
                  onClick={() => setShowShaHelp(true)}
                  className="mt-2 text-rose-300 underline font-semibold flex items-center gap-1 hover:text-white"
                >
                  <Info className="w-3.5 h-3.5" /> Hướng dẫn chẩn đoán SHA-1 & Client ID
                </button>
              )}
            </div>
          </div>
        )}

        {/* Danh sách hành động đăng nhập */}
        <div className="space-y-3 relative z-10">
          <button
            type="button"
            onClick={handleSignIn}
            disabled={isLoading}
            className="w-full py-3.5 px-4 bg-white hover:bg-slate-100 text-slate-900 font-bold rounded-2xl flex items-center justify-center gap-3 transition-all duration-150 shadow-md active:scale-[0.98] disabled:opacity-50 cursor-pointer"
          >
            {isLoading ? (
              <RefreshCw className="w-5 h-5 animate-spin text-slate-700" />
            ) : (
              <svg className="w-5 h-5" viewBox="0 0 24 24">
                <path
                  fill="#4285F4"
                  d="M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92c-.26 1.37-1.04 2.53-2.21 3.31v2.77h3.57c2.08-1.92 3.28-4.74 3.28-8.09z"
                />
                <path
                  fill="#34A853"
                  d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z"
                />
                <path
                  fill="#FBBC05"
                  d="M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.06H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.94l2.85-2.22.81-.63z"
                />
                <path
                  fill="#EA4335"
                  d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.06l3.66 2.84c.87-2.6 3.3-4.52 6.16-4.52z"
                />
              </svg>
            )}
            <span>{isLoading ? 'Đang xác thực...' : 'Đăng nhập với Google'}</span>
          </button>

          <button
            type="button"
            onClick={handleOfflineMode}
            disabled={isLoading}
            className="w-full py-3 px-4 bg-slate-800/80 hover:bg-slate-800 text-slate-300 font-semibold text-xs rounded-2xl flex items-center justify-center gap-2 border border-slate-700/60 transition-colors active:scale-[0.98] cursor-pointer"
          >
            <HardDrive className="w-4 h-4 text-emerald-400" />
            <span>Sử dụng Ngoại Tuyến (Offline Mode)</span>
          </button>

          {/* Thanh phiên bản & Kiểm tra cập nhật APK */}
          <div className="pt-3 px-1 flex items-center justify-between text-xs text-slate-400 border-t border-slate-800/60 mt-3">
            <span className="flex items-center gap-1.5 font-mono text-[11px]">
              <span className="w-1.5 h-1.5 rounded-full bg-emerald-500 animate-pulse" />
              v{currentVersion}
            </span>

            {hasNewUpdate ? (
              <button
                type="button"
                onClick={onOpenUpdateModal}
                className="flex items-center gap-1 text-[11px] font-bold text-emerald-400 hover:text-emerald-300 bg-emerald-500/10 px-2 py-0.5 rounded-full border border-emerald-500/30 animate-pulse cursor-pointer"
              >
                <ArrowUpCircle className="w-3.5 h-3.5" />
                Có bản v{newVersion} • Tải ngay
              </button>
            ) : onCheckUpdate ? (
              <button
                type="button"
                onClick={onCheckUpdate}
                className="flex items-center gap-1 text-[11px] text-slate-400 hover:text-slate-200 transition-colors cursor-pointer"
              >
                <RefreshCw className="w-3 h-3" />
                Kiểm tra cập nhật
              </button>
            ) : null}
          </div>
        </div>

        {/* Modal Hướng Dẫn SHA-1 & Google Client ID */}
        {showShaHelp && (
          <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/85 backdrop-blur-sm animate-in fade-in">
            <div className="w-full max-w-sm bg-slate-900 border border-slate-700 rounded-2xl p-5 text-slate-200 shadow-2xl relative">
              <button
                type="button"
                onClick={() => setShowShaHelp(false)}
                className="absolute right-4 top-4 p-1.5 rounded-full bg-slate-800 hover:bg-slate-700 text-slate-400 hover:text-white"
              >
                <X className="w-4 h-4" />
              </button>
              <h3 className="font-bold text-base text-white mb-2 flex items-center gap-2">
                <ShieldCheck className="w-5 h-5 text-emerald-400" />
                Thông Số Chẩn Đoán Android
              </h3>
              <p className="text-xs text-slate-400 mb-4 leading-relaxed">
                Nếu gặp lỗi Developer Error (Mã 10), hãy đối chiếu các thông số sau trong Google Cloud Console / Firebase Console:
              </p>

              <div className="space-y-3 text-xs">
                <div>
                  <span className="text-[11px] text-slate-400 font-semibold block mb-1">Package Name:</span>
                  <div className="flex items-center justify-between bg-slate-950 p-2.5 rounded-lg border border-slate-800">
                    <code className="text-emerald-400 text-[11px] truncate mr-2">{androidPackageName}</code>
                    <button
                      type="button"
                      onClick={() => handleCopy(androidPackageName, 'pkg')}
                      className="text-slate-400 hover:text-white shrink-0 p-1"
                    >
                      {copiedKey === 'pkg' ? <Check className="w-3.5 h-3.5 text-emerald-400" /> : <Copy className="w-3.5 h-3.5" />}
                    </button>
                  </div>
                </div>

                <div>
                  <span className="text-[11px] text-slate-400 font-semibold block mb-1">Cố định SHA-1 Fingerprint:</span>
                  <div className="flex items-center justify-between bg-slate-950 p-2.5 rounded-lg border border-slate-800">
                    <code className="text-amber-400 text-[10px] break-all mr-2">{fixedSha1}</code>
                    <button
                      type="button"
                      onClick={() => handleCopy(fixedSha1, 'sha1')}
                      className="text-slate-400 hover:text-white shrink-0 p-1"
                    >
                      {copiedKey === 'sha1' ? <Check className="w-3.5 h-3.5 text-emerald-400" /> : <Copy className="w-3.5 h-3.5" />}
                    </button>
                  </div>
                </div>

                <div>
                  <span className="text-[11px] text-slate-400 font-semibold block mb-1">Web Client ID:</span>
                  <div className="flex items-center justify-between bg-slate-950 p-2.5 rounded-lg border border-slate-800">
                    <code className="text-blue-400 text-[10px] truncate mr-2">{serverClientId}</code>
                    <button
                      type="button"
                      onClick={() => handleCopy(serverClientId, 'client')}
                      className="text-slate-400 hover:text-white shrink-0 p-1"
                    >
                      {copiedKey === 'client' ? <Check className="w-3.5 h-3.5 text-emerald-400" /> : <Copy className="w-3.5 h-3.5" />}
                    </button>
                  </div>
                </div>
              </div>

              <div className="mt-5">
                <button
                  type="button"
                  onClick={() => setShowShaHelp(false)}
                  className="w-full py-2.5 bg-emerald-600 hover:bg-emerald-500 text-white font-bold rounded-xl text-xs transition-colors"
                >
                  Đã hiểu & Đóng
                </button>
              </div>
            </div>
          </div>
        )}
      </div>
    </div>
  );
};
```

Bước 2: Viết hàm `onGoogleSignIn` kết nối Native / Web:
```typescript
import { GoogleAuth } from '@codetrix-studio/capacitor-google-auth';
import { Capacitor } from '@capacitor/core';

export async function performGoogleSignIn() {
  if (Capacitor.isNativePlatform()) {
    const user = await GoogleAuth.signIn();
    return {
      user: {
        email: user.email,
        displayName: user.name || user.givenName,
        photoURL: user.imageUrl,
      },
      accessToken: user.authentication?.accessToken,
    };
  } else {
    // Trên Web: Gọi Firebase Auth hoặc Google OAuth Popup
    throw new Error('Đang chạy trên web, vui lòng cấu hình web auth popup.');
  }
}
```

Bước 3: Nhúng component vào `App.tsx`:
```tsx
import React, { useState } from 'react';
import { LoginModal, AuthUserProfile } from './components/LoginModal';
import { performGoogleSignIn } from './auth';

export function App() {
  const [user, setUser] = useState<AuthUserProfile | null>(null);

  if (!user) {
    return (
      <LoginModal
        isOpen={true}
        appName="Sổ Chi Tiêu Thông Minh"
        currentVersion="1.0.0"
        onLogin={(authUser) => setUser(authUser)}
        onGoogleSignIn={performGoogleSignIn}
        onCheckUpdate={() => console.log('Checking update...')}
      />
    );
  }

  return <div>Xin chào {user.name} ({user.isOffline ? 'Chế độ Ngoại Tuyến' : user.email})</div>;
}
```

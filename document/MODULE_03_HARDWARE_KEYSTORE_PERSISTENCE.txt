1. MÔ TẢ CHỨC NĂNG & MỤC ĐÍCH SỬ DỤNG
--------------------------------------------------------------------------------
- Giải quyết triệt để lỗi "mất cài đặt / văng phiên đăng nhập khi cài đặt bản APK mới đè lên":
  + Nhiều dòng máy Android (Xiaomi, Samsung, Oppo, Vivo) tự động dọn dẹp hoặc reset vùng nhớ WebView LocalStorage khi có bản cập nhật APK ngoài Google Play.
- Kiến trúc lưu trữ 3 tầng bất biến:
  + Tầng 1 (Cốt lõi): Lưu vào phần cứng mã hóa an toàn của Android (Android KeyStore / EncryptedSharedPreferences thông qua `capacitor-secure-storage-plugin`). Tầng này KHÔNG THỂ bị xóa khi nâng cấp ứng dụng.
  + Tầng 2: Lưu vào Android Native `SharedPreferences` thông qua `@capacitor/preferences`.
  + Tầng 3: Lưu vào `localStorage` của trình duyệt để nạp dữ liệu tức thì (Zero-flicker Synchronous UI).
- Cơ chế chống xung đột hydrate (State Hydration Guard - `isRestored` Flag):
  + Ngăn chặn hiện tượng React State ghi đè giá trị mặc định (`false`, `null`) vào bộ nhớ trước khi kịp đọc dữ liệu cũ từ bộ nhớ phần cứng.
- Cơ chế ghi tức thì (Write-Through):
  + Khi người dùng bật/tắt thiết lập (như Khóa vân tay, Thông báo), hệ thống ghi thẳng vào cả 3 tầng ngay tại callback click, không phụ thuộc vào vòng đời render của component.

```ts
import { Capacitor } from '@capacitor/core';
import { SecureStoragePlugin } from 'capacitor-secure-storage-plugin';

export const isNativePlatform = (): boolean => {
  return Capacitor.isNativePlatform() || (typeof window !== 'undefined' && 'Capacitor' in window);
};

/**
 * Lưu dữ liệu an toàn vào chip phần cứng Android Keystore / iOS Keychain
 * Tự động fallback sang localStorage an toàn nếu chạy trên Web
 */
export async function setSecureItem(key: string, value: string): Promise<boolean> {
  try {
    if (isNativePlatform()) {
      await SecureStoragePlugin.set({ key, value });
      return true;
    } else {
      localStorage.setItem(`sec_${key}`, value);
      return true;
    }
  } catch (err) {
    console.warn(`[SecureStorage] Lỗi khi lưu khóa '${key}':`, err);
    try {
      localStorage.setItem(`sec_${key}`, value);
      return true;
    } catch {
      return false;
    }
  }
}

/**
 * Đọc dữ liệu từ chip bảo mật phần cứng
 */
export async function getSecureItem(key: string): Promise<string | null> {
  try {
    if (isNativePlatform()) {
      const res = await SecureStoragePlugin.get({ key });
      return res?.value ?? null;
    } else {
      return localStorage.getItem(`sec_${key}`);
    }
  } catch (err) {
    try {
      return localStorage.getItem(`sec_${key}`);
    } catch {
      return null;
    }
  }
}

/**
 * Xóa một khóa trong bộ nhớ bảo mật
 */
export async function removeSecureItem(key: string): Promise<boolean> {
  try {
    if (isNativePlatform()) {
      await SecureStoragePlugin.remove({ key });
    }
    localStorage.removeItem(`sec_${key}`);
    return true;
  } catch (err) {
    localStorage.removeItem(`sec_${key}`);
    return false;
  }
}

/**
 * Xóa toàn bộ dữ liệu trong bộ nhớ bảo mật (khi người dùng chủ động bấm Xóa Dữ Liệu)
 */
export async function clearAllSecureItems(): Promise<boolean> {
  try {
    if (isNativePlatform()) {
      await SecureStoragePlugin.clear();
    }
    return true;
  } catch {
    return false;
  }
}
```

File B: src/hooks/usePersistentState.ts (Hook bảo toàn dữ liệu 3 tầng)
--------------------------------------------------
```ts
import { useState, useEffect, useCallback, useRef } from 'react';
import { Preferences } from '@capacitor/preferences';
import { getSecureItem, setSecureItem } from '../utils/secureStorage';

export interface PersistentStateOptions<T> {
  key: string;
  defaultValue: T;
  serialize?: (val: T) => string;
  deserialize?: (raw: string) => T;
}

/**
 * Hook quản lý State có khả năng sống sót 100% qua mọi lần cập nhật đè APK
 */
export function usePersistentState<T>({
  key,
  defaultValue,
  serialize = (val: T) => JSON.stringify(val),
  deserialize = (raw: string) => JSON.parse(raw),
}: PersistentStateOptions<T>) {
  // 1. Khởi tạo đồng bộ từ localStorage (đảm bảo không bị giật/chớp giao diện)
  const [state, setState] = useState<T>(() => {
    try {
      const cached = localStorage.getItem(`persist_${key}`);
      if (cached !== null) {
        return deserialize(cached);
      }
    } catch {}
    return defaultValue;
  });

  const [isRestored, setIsRestored] = useState<boolean>(false);
  const stateRef = useRef<T>(state);
  stateRef.current = state;

  // 2. Khôi phục từ 2 tầng native (KeyStore & Preferences) khi khởi động
  useEffect(() => {
    let isMounted = true;

    const restoreFromNativeTiers = async () => {
      try {
        let restoredValue: T | null = null;

        // Ưu tiên 1: Đọc từ chip bảo mật phần cứng Android KeyStore
        const secureRaw = await getSecureItem(key);
        if (secureRaw !== null && secureRaw !== undefined && secureRaw !== '') {
          try {
            restoredValue = deserialize(secureRaw);
          } catch {}
        }

        // Ưu tiên 2: Đọc từ Android SharedPreferences (Preferences)
        if (restoredValue === null) {
          const { value: prefRaw } = await Preferences.get({ key: `pref_${key}` });
          if (prefRaw !== null && prefRaw !== undefined && prefRaw !== '') {
            try {
              restoredValue = deserialize(prefRaw);
            } catch {}
          }
        }

        // Nếu tìm thấy ở các tầng native, đồng bộ ngược vào State và localStorage
        if (isMounted && restoredValue !== null) {
          setState(restoredValue);
          try {
            localStorage.setItem(`persist_${key}`, serialize(restoredValue));
          } catch {}
        }
      } catch (err) {
        console.warn(`[usePersistentState] Lỗi khôi phục cho khóa '${key}':`, err);
      } finally {
        if (isMounted) {
          setIsRestored(true); // Đánh dấu hoàn thành quá trình khôi phục
        }
      }
    };

    restoreFromNativeTiers();

    return () => {
      isMounted = false;
    };
  }, [key]);

  // 3. Hàm cập nhật tức thì (Write-Through) xuống cả 3 tầng
  const updatePersistentState = useCallback(
    async (newValueOrUpdater: T | ((prev: T) => T)) => {
      setState((prev) => {
        const nextValue =
          typeof newValueOrUpdater === 'function'
            ? (newValueOrUpdater as (prev: T) => T)(prev)
            : newValueOrUpdater;

        const serialized = serialize(nextValue);

        // Tầng 3: Ghi tức thì vào localStorage
        try {
          localStorage.setItem(`persist_${key}`, serialized);
        } catch {}

        // Tầng 2: Ghi vào Android SharedPreferences
        Preferences.set({ key: `pref_${key}`, value: serialized }).catch(() => {});

        // Tầng 1: Ghi vào Android KeyStore phần cứng
        setSecureItem(key, serialized).catch(() => {});

        return nextValue;
      });
    },
    [key, serialize]
  );

  return [state, updatePersistentState, isRestored] as const;
}
```

### BƯỚC 3: Sử dụng như useState thông thường
```tsx
import React from 'react';
import { usePersistentState } from './hooks/usePersistentState';

export default function SettingsScreen() {
  // Thay thế useState bằng usePersistentState
  const [enableBiometric, setEnableBiometric, isRestored] = usePersistentState<boolean>({
    key: 'app_setting_biometrics',
    defaultValue: false,
  });

  const [notificationEnabled, setNotificationEnabled] = usePersistentState<boolean>({
    key: 'app_setting_notifications',
    defaultValue: true,
  });

  if (!isRestored) {
    return <div>Đang nạp cài đặt an toàn từ thiết bị...</div>;
  }

  return (
    <div className="p-4 space-y-4">
      <h2>Cài Đặt Ứng Dụng</h2>

      <label className="flex items-center space-x-2">
        <input
          type="checkbox"
          checked={enableBiometric}
          onChange={(e) => setEnableBiometric(e.target.checked)}
        />
        <span>Bật khóa vân tay (Được lưu trong Keystore phần cứng)</span>
      </label>

      <label className="flex items-center space-x-2">
        <input
          type="checkbox"
          checked={notificationEnabled}
          onChange={(e) => setNotificationEnabled(e.target.checked)}
        />
        <span>Bật thông báo đẩy</span>
      </label>
    </div>
  );
}
```
Mọi giá trị thay đổi sẽ được ghi đồng thời vào chip KeyStore và SharedPreferences của Android. Khi cập nhật đè file APK mới, thiết lập này sẽ tồn tại vĩnh viễn!

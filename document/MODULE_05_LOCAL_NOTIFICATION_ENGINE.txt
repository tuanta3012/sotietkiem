--------------------------------------------------------------------------------
1. MÔ TẢ CHỨC NĂNG & MỤC ĐÍCH SỬ DỤNG
--------------------------------------------------------------------------------
- Lập lịch và phát thông báo nhắc nhở trực tiếp từ phần cứng điện thoại Android / iOS mà KHÔNG cần máy chủ Backend (Serverless Local Notifications).
- Tự động cấu hình Notification Channel chuẩn Android 8.0+ (Oreo, Pie, Android 10-14):
  + Độ ưu tiên cao (`Importance.High`).
  + Bật rung, đèn LED báo và chuông thông báo mặc định của hệ điều hành.
- Khả năng kiểm tra và yêu cầu quyền gửi thông báo (Android 13+ `POST_NOTIFICATIONS` runtime permission).
- Lập lịch thông báo chính xác theo ngày giờ trong tương lai (ví dụ: nhắc hạn thanh toán, nhắc công việc, nhắc uống nước, sinh nhật).
- Hỗ trợ gửi thông báo tức thì (Test Notification) và xóa/hủy thông báo cũ theo danh sách ID.
- Fallback thông minh: Tự động dùng Web Notification API nếu chạy trên trình duyệt máy tính.

```ts
import { Capacitor } from '@capacitor/core';
import { LocalNotifications, ScheduleOptions, Channel } from '@capacitor/local-notifications';

export interface LocalNotificationPayload {
  id: number; // Số nguyên 32-bit duy nhất
  title: string;
  body: string;
  scheduleAt?: Date; // Thời điểm phát thông báo (nếu không truyền sẽ phát ngay lập tức)
  sound?: string;
  extraData?: Record<string, any>;
}

export const NotificationEngine = {
  /**
   * Kiểm tra xem nền tảng hiện tại có phải là native Android/iOS không
   */
  isNative(): boolean {
    return Capacitor.isNativePlatform();
  },

  /**
   * Khởi tạo kênh thông báo cho Android (Bắt buộc từ Android 8.0 trở lên)
   */
  async setupAndroidChannel(channelId: string = 'app_reminders', channelName: string = 'Nhắc Nhở Ứng Dụng'): Promise<void> {
    if (this.isNative()) {
      try {
        const channel: Channel = {
          id: channelId,
          name: channelName,
          description: 'Kênh phát thông báo nhắc nhở và cập nhật quan trọng',
          importance: 5, // High Importance (hiển thị popup trên thanh trạng thái)
          visibility: 1, // Public trên màn hình khóa
          vibration: true,
          lights: true,
          lightColor: '#10B981', // Màu xanh ngọc Emerald
        };
        await LocalNotifications.createChannel(channel);
      } catch (err) {
        console.warn('[NotificationEngine] Lỗi tạo Android Channel:', err);
      }
    }
  },

  /**
   * Kiểm tra trạng thái cấp quyền hiện tại
   */
  async checkPermission(): Promise<boolean> {
    try {
      if (this.isNative()) {
        const res = await LocalNotifications.checkPermissions();
        return res.display === 'granted';
      } else {
        if (typeof window !== 'undefined' && 'Notification' in window) {
          return Notification.permission === 'granted';
        }
      }
    } catch (err) {
      console.warn('[NotificationEngine] Lỗi checkPermission:', err);
    }
    return false;
  },

  /**
   * Yêu cầu người dùng cấp quyền thông báo
   */
  async requestPermission(): Promise<boolean> {
    try {
      if (this.isNative()) {
        const check = await LocalNotifications.checkPermissions();
        if (check.display === 'granted') return true;

        const req = await LocalNotifications.requestPermissions();
        if (req.display === 'granted') {
          await this.setupAndroidChannel();
          return true;
        }
        return false;
      } else {
        if (typeof window !== 'undefined' && 'Notification' in window) {
          if (Notification.permission === 'granted') return true;
          if (Notification.permission !== 'denied') {
            const res = await Notification.requestPermission();
            return res === 'granted';
          }
        }
      }
    } catch (err) {
      console.warn('[NotificationEngine] Lỗi requestPermission:', err);
    }
    return false;
  },

  /**
   * Lập lịch hoặc phát thông báo ngay lập tức
   */
  async schedule(notification: LocalNotificationPayload, channelId: string = 'app_reminders'): Promise<boolean> {
    const hasPermission = await this.checkPermission();
    if (!hasPermission) {
      const granted = await this.requestPermission();
      if (!granted) return false;
    }

    try {
      if (this.isNative()) {
        await this.setupAndroidChannel(channelId);

        const options: ScheduleOptions = {
          notifications: [
            {
              id: notification.id,
              title: notification.title,
              body: notification.body,
              schedule: notification.scheduleAt ? { at: notification.scheduleAt } : undefined,
              channelId: channelId,
              smallIcon: 'ic_stat_icon_config_sample', // Icon mặc định trong Android res/drawable
              iconColor: '#10B981',
              extra: notification.extraData || null,
            },
          ],
        };

        await LocalNotifications.schedule(options);
        return true;
      } else {
        // Fallback Web Browser Notification
        if (typeof window !== 'undefined' && 'Notification' in window && Notification.permission === 'granted') {
          if (notification.scheduleAt) {
            const delayMs = notification.scheduleAt.getTime() - Date.now();
            if (delayMs > 0) {
              setTimeout(() => {
                new Notification(notification.title, { body: notification.body });
              }, delayMs);
            }
          } else {
            new Notification(notification.title, { body: notification.body });
          }
          return true;
        }
      }
    } catch (err) {
      console.warn('[NotificationEngine] Lỗi schedule:', err);
    }
    return false;
  },

  /**
   * Hủy một hoặc nhiều thông báo đã lập lịch
   */
  async cancel(notificationIds: number[]): Promise<void> {
    if (this.isNative() && notificationIds.length > 0) {
      try {
        await LocalNotifications.cancel({
          notifications: notificationIds.map((id) => ({ id })),
        });
      } catch (err) {
        console.warn('[NotificationEngine] Lỗi cancel:', err);
      }
    }
  },

  /**
   * Xóa toàn bộ thông báo đang chờ phát
   */
  async cancelAllPending(): Promise<void> {
    if (this.isNative()) {
      try {
        const pending = await LocalNotifications.getPending();
        if (pending.notifications.length > 0) {
          await LocalNotifications.cancel({
            notifications: pending.notifications.map((n) => ({ id: n.id })),
          });
        }
      } catch (err) {
        console.warn('[NotificationEngine] Lỗi cancelAllPending:', err);
      }
    }
  },
};
```

### BƯỚC 3: Lập lịch thông báo ngay lập tức hoặc theo hẹn giờ
```tsx
import React from 'react';
import { NotificationEngine } from './utils/notificationEngine';

export default function NotificationDemo() {
  // 1. Phát thông báo thử nghiệm ngay lập tức
  const handleTestNow = async () => {
    const success = await NotificationEngine.schedule({
      id: Math.floor(Date.now() % 100000),
      title: 'Chào mừng bạn!',
      body: 'Đây là thông báo native không cần máy chủ backend.',
    });
    if (!success) alert('Vui lòng cấp quyền thông báo trên điện thoại!');
  };

  // 2. Lập lịch thông báo sau 10 giây (hoặc vào 8h00 sáng mai)
  const handleScheduleIn10Seconds = async () => {
    const triggerTime = new Date(Date.now() + 10 * 1000); // 10 giây sau

    await NotificationEngine.schedule({
      id: 1001,
      title: 'Đã đến giờ nhắc việc!',
      body: 'Kiểm tra tiến độ công việc trong ngày hôm nay.',
      scheduleAt: triggerTime,
    });
    alert('Đã lên lịch thành công! Khóa màn hình để kiểm tra sau 10 giây.');
  };

  return (
    <div className="p-4 space-x-2">
      <button onClick={handleTestNow} className="px-4 py-2 bg-blue-600 text-white rounded-lg">
        Bắn thông báo ngay
      </button>

      <button onClick={handleScheduleIn10Seconds} className="px-4 py-2 bg-emerald-600 text-white rounded-lg">
        Hẹn giờ sau 10 giây
      </button>
    </div>
  );
}
```
Thông báo sẽ tự động phát bằng âm thanh và rung của điện thoại ngay cả khi ứng dụng bị đóng hoặc màn hình đang tắt!

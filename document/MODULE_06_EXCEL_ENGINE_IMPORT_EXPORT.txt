1. MÔ TẢ CHỨC NĂNG & MỤC ĐÍCH SỬ DỤNG
--------------------------------------------------------------------------------
Module này cung cấp một bộ công cụ xử lý tệp tin bảng tính Excel (.xlsx) cấp độ
sản xuất (Production-grade), giải quyết các bài toán phổ biến trong ứng dụng:

- Tải động (Dynamic Import) thư viện SheetJS (xlsx): Chỉ tải thư viện khi người
  dùng thực sự bấm Đọc hoặc Xuất Excel, giúp kích thước Bundle ban đầu của ứng dụng
  siêu nhẹ (giảm hơn 500KB bundle size ban đầu).
- Đọc tệp tin Excel (Import): Đọc File từ máy tính, bộ nhớ điện thoại hoặc mảng
  dữ liệu ArrayBuffer/Base64, tự động phát hiện hàng tiêu đề, chuẩn hóa định dạng số
  Việt Nam (dấu chấm/phẩy), ngày tháng và xuất ra mảng Object có kiểu dữ liệu (Typesafe).
- Xuất tệp tin Excel (Export): Tạo tệp tin .xlsx chuyên nghiệp với nhiều Sheet,
  tự động tính toán độ rộng cột (Column Widths), định dạng tiêu đề.
- Cơ chế Tải về Đa Nền tảng (Cross-Platform Download):
  + Trên Web: Tự động kích hoạt luồng tải tệp tin của trình duyệt (Browser Download).
  + Trên Mobile (Android/iOS): Ghi tệp Base64 vào bộ nhớ đệm Cache thông qua
    @capacitor/filesystem, sau đó kích hoạt hộp thoại Native Share của Android (@capacitor/share)
    để người dùng có thể mở ngay bằng Excel, Google Sheets, Zalo, Gmail, Drive...

3. MÃ NGUỒN HOÀN CHỈNH (FULL SOURCE CODE)
--------------------------------------------------------------------------------
Lưu tệp này tại: /src/utils/excelEngine.ts

```typescript
import { Capacitor } from '@capacitor/core';
import { Filesystem, Directory } from '@capacitor/filesystem';
import { Share } from '@capacitor/share';

// 1. Tải động module XLSX khi cần và cache lại để tối ưu dung lượng bundle ban đầu
let xlsxModule: typeof import('xlsx') | null = null;
export const getXLSX = async () => {
  if (!xlsxModule) {
    xlsxModule = await import('xlsx');
  }
  return xlsxModule;
};

export interface ColumnDefinition {
  header: string;
  key: string;
  width?: number; // Độ rộng ký tự (wch)
  formatter?: (val: any) => string | number;
}

export interface SheetDefinition<T = any> {
  sheetName: string;
  columns: ColumnDefinition[];
  data: T[];
}

/**
 * 2. Đọc tệp tin Excel từ File object hoặc ArrayBuffer thành mảng 2 chiều (Matrix)
 */
export async function parseExcelToMatrix(fileOrBuffer: File | ArrayBuffer): Promise<{
  sheets: Record<string, any[][]>;
  firstSheetData: any[][];
}> {
  const XLSX = await getXLSX();
  let arrayBuffer: ArrayBuffer;

  if (fileOrBuffer instanceof File) {
    arrayBuffer = await fileOrBuffer.arrayBuffer();
  } else {
    arrayBuffer = fileOrBuffer;
  }

  const workbook = XLSX.read(arrayBuffer, { type: 'array' });
  const result: Record<string, any[][]> = {};
  let firstSheetName = workbook.SheetNames[0] || '';

  for (const sheetName of workbook.SheetNames) {
    const ws = workbook.Sheets[sheetName];
    if (ws) {
      result[sheetName] = XLSX.utils.sheet_to_json(ws, { header: 1, defval: '' }) as any[][];
    }
  }

  return {
    sheets: result,
    firstSheetData: result[firstSheetName] || [],
  };
}

/**
 * 3. Chuyển đổi dữ liệu đối tượng (Array of Objects) thành Workbook có format
 */
export async function buildExcelWorkbook(sheetConfigs: SheetDefinition[]): Promise<any> {
  const XLSX = await getXLSX();
  const workbook = XLSX.utils.book_new();

  for (const sheet of sheetConfigs) {
    // Tạo ma trận dữ liệu: Hàng 0 là tiêu đề
    const matrix: (string | number)[][] = [];
    const headers = sheet.columns.map((c) => c.header);
    matrix.push(headers);

    for (const item of sheet.data) {
      const row = sheet.columns.map((col) => {
        const val = (item as any)[col.key];
        if (col.formatter) {
          return col.formatter(val);
        }
        return val !== undefined && val !== null ? val : '';
      });
      matrix.push(row);
    }

    const worksheet = XLSX.utils.aoa_to_sheet(matrix);

    // Cài đặt độ rộng cột
    worksheet['!cols'] = sheet.columns.map((col) => ({
      wch: col.width || Math.max(col.header.length + 4, 12),
    }));

    XLSX.utils.book_append_sheet(workbook, worksheet, sheet.sheetName.slice(0, 31)); // Excel giới hạn tên sheet 31 ký tự
  }

  return workbook;
}

/**
 * 4. Xuất mảng dữ liệu ra tệp .xlsx và tải về trên cả Web và Android/iOS Native
 */
export async function exportToExcelFile(
  sheetConfigs: SheetDefinition[],
  fileName = 'Du_Lieu_Xuat.xlsx'
): Promise<void> {
  const XLSX = await getXLSX();
  const workbook = await buildExcelWorkbook(sheetConfigs);

  if (Capacitor.isNativePlatform()) {
    // Trên nền tảng di động Android / iOS:
    try {
      const base64Data = XLSX.write(workbook, { bookType: 'xlsx', type: 'base64' });
      
      const writeResult = await Filesystem.writeFile({
        path: fileName,
        data: base64Data,
        directory: Directory.Cache,
      });

      // Mở hộp thoại Share của Native OS để gửi qua Zalo, Drive, Gmail hoặc mở bằng Excel
      await Share.share({
        title: 'Chia sẻ tệp Excel',
        text: `Xuất tệp dữ liệu: ${fileName}`,
        url: writeResult.uri,
      });
    } catch (err) {
      console.error('[ExcelEngine] Lỗi xuất tệp trên mobile native:', err);
      throw new Error('Không thể chia sẻ tệp Excel trên thiết bị: ' + (err instanceof Error ? err.message : String(err)));
    }
  } else {
    // Trên trình duyệt Web máy tính / điện thoại
    XLSX.writeFile(workbook, fileName);
  }
}

/**
 * 5. Tiện ích chuẩn hóa số tiếng Việt (VD: "1.500.000,50" -> 1500000.5)
 */
export function parseVietnameseNumber(val: any): number {
  if (typeof val === 'number') return isNaN(val) ? 0 : val;
  if (!val) return 0;
  let str = String(val).trim().replace(/[^\d.,\-]/g, '');
  if (!str) return 0;

  // Trường hợp chuẩn VN: dấu chấm phân tách hàng nghìn (1.000.000)
  if (/^\d{1,3}(\.\d{3})+$/.test(str)) {
    str = str.replace(/\./g, '');
  } else if (str.includes('.') && str.includes(',')) {
    str = str.replace(/\./g, '').replace(',', '.');
  } else if (str.includes(',')) {
    str = str.replace(',', '.');
  }
  const res = parseFloat(str);
  return isNaN(res) ? 0 : res;
}
```

4. HƯỚNG DẪN TÍCH HỢP VÀO APP MỚI CHỈ TRONG 3 BƯỚC
--------------------------------------------------------------------------------
Bước 1: Copy file trên vào thư mục `/src/utils/excelEngine.ts`.
Bước 2: Cài đặt thư viện: `npm install xlsx @capacitor/filesystem @capacitor/share`
Bước 3: Sử dụng trực tiếp trong Component React của bạn:

```tsx
import React from 'react';
import { exportToExcelFile, parseExcelToMatrix } from './utils/excelEngine';

export function ExportDemo() {
  const sampleData = [
    { id: '001', name: 'Nguyễn Văn A', amount: 15000000, date: '2026-09-29' },
    { id: '002', name: 'Trần Thị B', amount: 32000000, date: '2026-09-30' },
  ];

  const handleExport = async () => {
    await exportToExcelFile([
      {
        sheetName: 'DanhSach',
        columns: [
          { header: 'Mã số', key: 'id', width: 10 },
          { header: 'Họ và tên', key: 'name', width: 22 },
          { 
            header: 'Số tiền (VNĐ)', 
            key: 'amount', 
            width: 18,
            formatter: (v) => Number(v).toLocaleString('vi-VN')
          },
          { header: 'Ngày tạo', key: 'date', width: 15 },
        ],
        data: sampleData,
      }
    ], 'Bao_Cao_Nhan_Su_2026.xlsx');
  };

  const handleImport = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const { firstSheetData } = await parseExcelToMatrix(file);
    console.log('Dữ liệu ma trận Excel đã đọc:', firstSheetData);
  };

  return (
    <div className="p-4 space-x-3">
      <button onClick={handleExport} className="px-4 py-2 bg-emerald-600 text-white rounded-xl font-bold">
        Xuất file Excel
      </button>
      <input type="file" accept=".xlsx, .xls" onChange={handleImport} />
    </div>
  );
}
```

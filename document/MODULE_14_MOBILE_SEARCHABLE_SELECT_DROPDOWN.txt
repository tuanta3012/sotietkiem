1. MÔ TẢ CHỨC NĂNG & MỤC ĐÍCH SỬ DỤNG
--------------------------------------------------------------------------------
- Giải quyết bài toán chọn lựa trên màn hình cảm ứng di động (Mobile Touch UX):
  + Khắc phục nhược điểm của thẻ `<select>` mặc định của trình duyệt (giao diện thô, không tìm kiếm được, khó tùy biến icon/badge).
  + Tìm kiếm tức thì (Instant Search Filter) theo từ khóa không dấu hoặc có dấu.
  + Tối ưu chống nảy bàn phím trên điện thoại di động: Chỉ focus ô tìm kiếm khi người dùng chủ động chạm vào, không tự động giật màn hình khi mở menu.
  + Phân nhóm & Tab lựa chọn: Hỗ trợ chia danh sách thành các tab (ví dụ: "Tất cả" và "Thường dùng / Yêu thích").
  + Tự động định vị thông minh: Chống tràn mép màn hình điện thoại (`alignRight`, `dropdownFullWidth`, responsive max-height).
  + Xử lý sự kiện click ngoài (`mousedown`, `touchstart`) và phím Escape để tự đóng dropdown.
  + Hỗ trợ hiển thị đa dạng: Icon vector, hình ảnh logo, tên chính, tên phụ (subLabel), huy hiệu đếm số lượng, dấu tích chọn (active indicator).

```tsx
import React, { useState, useRef, useEffect, useMemo } from 'react';
import { ChevronDown, Search, Check, X, Star } from 'lucide-react';

export interface SelectItem<T = string> {
  id: T;
  label: string;
  subLabel?: string;
  badge?: string | number;
  icon?: React.ReactNode;
  isFavorite?: boolean;
}

export interface SearchableSelectProps<T = string> {
  value: T;
  onChange: (value: T) => void;
  items: SelectItem<T>[];
  placeholder?: string;
  searchPlaceholder?: string;
  enableSearch?: boolean;
  enableTabs?: boolean;
  tabAllLabel?: string;
  tabFavoriteLabel?: string;
  alignRight?: boolean;
  disabled?: boolean;
  className?: string;
}

export function SearchableSelect<T = string>({
  value,
  onChange,
  items,
  placeholder = 'Chọn một mục...',
  searchPlaceholder = 'Tìm kiếm nhanh...',
  enableSearch = true,
  enableTabs = false,
  tabAllLabel = 'Tất cả',
  tabFavoriteLabel = 'Thường dùng',
  alignRight = false,
  disabled = false,
  className = '',
}: SearchableSelectProps<T>) {
  const [isOpen, setIsOpen] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [activeTab, setActiveTab] = useState<'all' | 'favorite'>('all');

  const containerRef = useRef<HTMLDivElement>(null);

  // Tìm item hiện tại đang được chọn
  const selectedItem = useMemo(() => {
    return items.find((item) => item.id === value);
  }, [items, value]);

  // Click ngoài đóng dropdown
  useEffect(() => {
    const handleClickOutside = (e: MouseEvent | TouchEvent) => {
      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
        setIsOpen(false);
      }
    };
    if (isOpen) {
      document.addEventListener('mousedown', handleClickOutside);
      document.addEventListener('touchstart', handleClickOutside);
    }
    return () => {
      document.removeEventListener('mousedown', handleClickOutside);
      document.removeEventListener('touchstart', handleClickOutside);
    };
  }, [isOpen]);

  // Reset từ khóa khi đóng
  useEffect(() => {
    if (!isOpen) {
      setSearchQuery('');
    }
  }, [isOpen]);

  // Lọc danh sách theo tab và từ khóa
  const filteredItems = useMemo(() => {
    let list = items;
    if (enableTabs && activeTab === 'favorite') {
      list = list.filter((item) => item.isFavorite);
    }
    const q = searchQuery.trim().toLowerCase();
    if (!q) return list;
    return list.filter(
      (item) =>
        item.label.toLowerCase().includes(q) ||
        (item.subLabel && item.subLabel.toLowerCase().includes(q))
    );
  }, [items, searchQuery, activeTab, enableTabs]);

  return (
    <div ref={containerRef} className={`relative w-full ${className}`}>
      {/* Nút bấm hiển thị chính */}
      <button
        type="button"
        disabled={disabled}
        onClick={() => !disabled && setIsOpen(!isOpen)}
        className={`w-full bg-white dark:bg-slate-900 border border-slate-300 dark:border-slate-700 text-slate-800 dark:text-slate-100 flex items-center justify-between text-left transition-all duration-150 py-2.5 px-3.5 text-xs font-semibold rounded-2xl focus:outline-none focus:ring-2 focus:ring-emerald-500/30 focus:border-emerald-500 ${
          disabled
            ? 'opacity-50 cursor-not-allowed bg-slate-50 dark:bg-slate-800'
            : 'cursor-pointer hover:border-slate-400 dark:hover:border-slate-600 shadow-xs'
        } ${isOpen ? 'ring-2 ring-emerald-500/30 border-emerald-500' : ''}`}
      >
        <div className="truncate flex items-center gap-2 pr-2">
          {selectedItem?.icon}
          <div className="truncate">
            <span className="truncate">{selectedItem ? selectedItem.label : placeholder}</span>
            {selectedItem?.subLabel && (
              <span className="text-[10px] text-slate-400 font-normal ml-1.5 hidden sm:inline">
                ({selectedItem.subLabel})
              </span>
            )}
          </div>
        </div>
        <ChevronDown
          className={`w-4 h-4 text-slate-400 shrink-0 transition-transform duration-200 ${
            isOpen ? 'rotate-180 text-emerald-600' : ''
          }`}
        />
      </button>

      {/* Menu thả xuống */}
      {isOpen && (
        <div
          className={`absolute mt-1.5 bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-800 rounded-2xl shadow-2xl z-50 overflow-hidden flex flex-col animate-in fade-in zoom-in-95 duration-100 ${
            alignRight ? 'right-0 w-72 sm:w-80' : 'left-0 right-0 w-full sm:w-80'
          }`}
          style={{ maxHeight: '380px' }}
        >
          {/* Ô tìm kiếm */}
          {enableSearch && (
            <div className="p-2.5 border-b border-slate-100 dark:border-slate-800 bg-slate-50/50 dark:bg-slate-950/40">
              <div className="relative">
                <Search className="w-3.5 h-3.5 absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
                <input
                  type="text"
                  value={searchQuery}
                  onChange={(e) => setSearchQuery(e.target.value)}
                  placeholder={searchPlaceholder}
                  className="w-full pl-8 pr-7 py-2 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl text-xs text-slate-800 dark:text-slate-100 focus:outline-none focus:border-emerald-500 focus:ring-1 focus:ring-emerald-500"
                />
                {searchQuery && (
                  <button
                    type="button"
                    onClick={() => setSearchQuery('')}
                    className="absolute right-2.5 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600 dark:hover:text-slate-200 p-0.5"
                  >
                    <X className="w-3 h-3" />
                  </button>
                )}
              </div>
            </div>
          )}

          {/* Tab phân loại (Tất cả / Yêu thích) */}
          {enableTabs && (
            <div className="flex border-b border-slate-100 dark:border-slate-800 px-2 pt-1.5 bg-slate-50/30 dark:bg-slate-950/20">
              <button
                type="button"
                onClick={() => setActiveTab('all')}
                className={`flex-1 py-1.5 text-[11px] font-bold text-center border-b-2 transition-colors cursor-pointer ${
                  activeTab === 'all'
                    ? 'border-emerald-500 text-emerald-600 dark:text-emerald-400'
                    : 'border-transparent text-slate-500 hover:text-slate-700'
                }`}
              >
                {tabAllLabel}
              </button>
              <button
                type="button"
                onClick={() => setActiveTab('favorite')}
                className={`flex-1 py-1.5 text-[11px] font-bold text-center border-b-2 transition-colors flex items-center justify-center gap-1 cursor-pointer ${
                  activeTab === 'favorite'
                    ? 'border-emerald-500 text-emerald-600 dark:text-emerald-400'
                    : 'border-transparent text-slate-500 hover:text-slate-700'
                }`}
              >
                <Star className="w-3 h-3 text-amber-400 fill-amber-400" />
                {tabFavoriteLabel}
              </button>
            </div>
          )}

          {/* Danh sách các tùy chọn */}
          <div className="overflow-y-auto flex-1 p-1 divide-y divide-slate-50 dark:divide-slate-800/50">
            {filteredItems.length === 0 ? (
              <div className="py-6 text-center text-xs text-slate-400">
                Không tìm thấy kết quả phù hợp
              </div>
            ) : (
              filteredItems.map((item) => {
                const isSelected = item.id === value;
                return (
                  <button
                    key={String(item.id)}
                    type="button"
                    onClick={() => {
                      onChange(item.id);
                      setIsOpen(false);
                    }}
                    className={`w-full text-left px-3 py-2.5 text-xs rounded-xl flex items-center justify-between transition-colors cursor-pointer ${
                      isSelected
                        ? 'bg-emerald-50 dark:bg-emerald-950/40 text-emerald-900 dark:text-emerald-200 font-bold'
                        : 'text-slate-700 dark:text-slate-300 hover:bg-slate-100/70 dark:hover:bg-slate-800 font-medium'
                    }`}
                  >
                    <div className="flex items-center gap-2.5 min-w-0 pr-2">
                      {item.icon}
                      <div className="truncate">
                        <div className="truncate flex items-center gap-1.5">
                          <span>{item.label}</span>
                          {item.badge && (
                            <span className="text-[10px] font-bold px-1.5 py-0.5 rounded-full bg-slate-100 dark:bg-slate-800 text-slate-600 dark:text-slate-400">
                              {item.badge}
                            </span>
                          )}
                        </div>
                        {item.subLabel && (
                          <div className="text-[10px] text-slate-400 font-normal truncate mt-0.5">
                            {item.subLabel}
                          </div>
                        )}
                      </div>
                    </div>
                    {isSelected && (
                      <div className="w-4 h-4 rounded-full bg-emerald-600 text-white flex items-center justify-center shrink-0">
                        <Check className="w-2.5 h-2.5 stroke-[3]" />
                      </div>
                    )}
                  </button>
                );
              })
            )}
          </div>
        </div>
      )}
    </div>
  );
}
```

```tsx
import React, { useState } from 'react';
import { SearchableSelect, SelectItem } from './components/SearchableSelect';
import { Building2, Wallet } from 'lucide-react';

const bankOptions: SelectItem[] = [
  { id: 'vcb', label: 'Vietcombank', subLabel: 'Ngân hàng Ngoại thương', isFavorite: true, icon: <Building2 className="w-4 h-4 text-emerald-600" /> },
  { id: 'tcb', label: 'Techcombank', subLabel: 'Ngân hàng Kỹ thương', isFavorite: true, icon: <Building2 className="w-4 h-4 text-rose-600" /> },
  { id: 'mbb', label: 'MB Bank', subLabel: 'Ngân hàng Quân đội', isFavorite: false, icon: <Building2 className="w-4 h-4 text-blue-600" /> },
];

export function FilterBar() {
  const [selectedBank, setSelectedBank] = useState('vcb');

  return (
    <div className="w-72">
      <SearchableSelect
        value={selectedBank}
        onChange={(val) => setSelectedBank(val)}
        items={bankOptions}
        enableSearch={true}
        enableTabs={true}
        searchPlaceholder="Gõ tên ngân hàng..."
      />
    </div>
  );
}
```

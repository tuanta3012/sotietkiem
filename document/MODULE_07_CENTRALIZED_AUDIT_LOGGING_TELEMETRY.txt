1. MÔ TẢ CHỨC NĂNG & MỤC ĐÍCH SỬ DỤNG
--------------------------------------------------------------------------------
Hệ thống Nhật ký Kiểm toán & Báo cáo Lỗi Tập trung (Audit Logging & Error Telemetry)
giải quyết bài toán giám sát, theo dõi sự cố trên ứng dụng di động:

- Lưu vết mọi thao tác quan trọng: Đăng nhập, thay đổi dữ liệu, đồng bộ đám mây,
  lỗi mạng, crash hệ thống...
- Lưu trữ 2 tầng linh hoạt:
  + Tầng Cục bộ (Local): Lưu trong localStorage của thiết bị với giới hạn quay vòng
    (Rolling Buffer) tối đa 50 bản ghi gần nhất để không làm đầy bộ nhớ máy.
  + Tầng Đám mây (Cloud Firestore): Tự động đẩy không đồng bộ (Non-blocking) lên
    bộ sưu tập `audit_logs` trên Firestore. Quản trị viên (Admin) có thể mở ứng dụng
    ở máy của mình và xem toàn bộ nhật ký lỗi/hoạt động của tất cả các máy trạm khác.
- Giao diện Xem Nhật ký Đẹp mắt (`AuditLogModal`):
  + Chuyển tab Đám mây (Cloud) / Máy này (Local).
  + Lọc theo trạng thái: Tất cả, Thành công (Success), Lỗi (Error), Cảnh báo (Warning).
  + Xem chi tiết payload JSON kỹ thuật, sao chép nhanh (Copy to clipboard).
  + Xuất file nhật ký .txt hoặc chia sẻ qua Native Share trên điện thoại.

```typescript
export interface AuditLogEntry {
  id: string;
  timestamp: number;
  timeStr: string; // VD: "29/09/2026 18:30:15"
  type: string;    // VD: 'USER_LOGIN', 'SYNC_SUCCESS', 'SYNC_ERROR', 'DATA_UPDATE'
  title: string;
  status: 'success' | 'warning' | 'error' | 'info';
  userEmail?: string;
  userName?: string;
  currentRole?: string;
  platform?: string; // 'Android' | 'Web' | 'iOS'
  appVersion?: string;
  summary: string;
  details?: Record<string, any>;
  errorMessage?: string;
}

const STORAGE_KEY = 'app_audit_log_v1';
const MAX_LOG_ENTRIES = 50;

/**
 * 1. Đọc danh sách log từ bộ nhớ cục bộ
 */
export function getLocalAuditLogs(): AuditLogEntry[] {
  try {
    if (typeof localStorage === 'undefined') return [];
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return [];
    const list = JSON.parse(raw);
    return Array.isArray(list) ? list : [];
  } catch (err) {
    console.warn('[AuditLogger] Lỗi đọc log từ storage:', err);
    return [];
  }
}

/**
 * 2. Xóa toàn bộ log trên máy này
 */
export function clearLocalAuditLogs(): void {
  try {
    if (typeof localStorage !== 'undefined') {
      localStorage.removeItem(STORAGE_KEY);
    }
  } catch (err) {
    console.warn('[AuditLogger] Lỗi xóa log:', err);
  }
}

/**
 * 3. Ghi nhận một sự kiện nhật ký mới
 */
export function recordAuditLog(
  entry: Omit<AuditLogEntry, 'id' | 'timestamp' | 'timeStr'>,
  onPushToCloud?: (log: AuditLogEntry) => Promise<void>
): AuditLogEntry {
  const now = new Date();
  const timeStr = now.toLocaleString('vi-VN', {
    day: '2-digit',
    month: '2-digit',
    year: 'numeric',
    hour: '2-digit',
    minute: '2-digit',
    second: '2-digit',
  });

  const fullEntry: AuditLogEntry = {
    ...entry,
    id: `log_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
    timestamp: Date.now(),
    timeStr,
    platform: typeof navigator !== 'undefined' ? (navigator.userAgent.includes('Android') ? 'Android' : 'Web') : 'Web',
  };

  // Lưu cục bộ (Rolling Buffer 50 mục)
  try {
    const existing = getLocalAuditLogs();
    const updated = [fullEntry, ...existing].slice(0, MAX_LOG_ENTRIES);
    if (typeof localStorage !== 'undefined') {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(updated));
    }
  } catch (err) {
    console.warn('[AuditLogger] Lỗi lưu log cục bộ:', err);
  }

  // Tùy chọn đẩy lên Firestore đám mây ngầm
  if (onPushToCloud) {
    try {
      onPushToCloud(fullEntry).catch(() => {});
    } catch {}
  }

  return fullEntry;
}

/**
 * 4. Xuất danh sách nhật ký thành văn bản Plain Text để tải về / chia sẻ
 */
export function formatAuditLogsAsText(logs: AuditLogEntry[]): string {
  const lines: string[] = [
    '================================================================',
    '                   NHẬT KÝ KIỂM TOÁN HỆ THỐNG                   ',
    ` Xuất lúc: ${new Date().toLocaleString('vi-VN')}                `,
    ` Tổng số bản ghi: ${logs.length}                                `,
    '================================================================\n',
  ];

  logs.forEach((log, index) => {
    lines.push(`[#${index + 1}] ${log.timeStr} | [${log.status.toUpperCase()}] | ${log.type}`);
    lines.push(`Tiêu đề: ${log.title}`);
    lines.push(`Tóm tắt: ${log.summary}`);
    if (log.userEmail) lines.push(`Người dùng: ${log.userEmail} (${log.currentRole || 'N/A'}) - Nền tảng: ${log.platform || 'N/A'}`);
    if (log.errorMessage) lines.push(`Chi tiết lỗi: ${log.errorMessage}`);
    if (log.details) lines.push(`Dữ liệu đính kèm: ${JSON.stringify(log.details, null, 2)}`);
    lines.push('----------------------------------------------------------------\n');
  });

  return lines.join('\n');
}
```

```tsx
import React, { useState, useEffect, useMemo } from 'react';
import {
  X,
  FileText,
  Copy,
  Download,
  Trash2,
  RefreshCw,
  CheckCircle2,
  AlertCircle,
  AlertTriangle,
  ChevronDown,
  ChevronUp,
  Cloud,
  Smartphone,
  Check,
} from 'lucide-react';
import {
  getLocalAuditLogs,
  clearLocalAuditLogs,
  formatAuditLogsAsText,
  AuditLogEntry,
} from '../utils/auditLogger';
import { Capacitor } from '@capacitor/core';
import { Filesystem, Directory, Encoding } from '@capacitor/filesystem';
import { Share } from '@capacitor/share';

interface AuditLogModalProps {
  isOpen: boolean;
  onClose: () => void;
  fetchCloudLogs?: () => Promise<AuditLogEntry[]>;
}

export const AuditLogModal: React.FC<AuditLogModalProps> = ({
  isOpen,
  onClose,
  fetchCloudLogs,
}) => {
  const [activeTab, setActiveTab] = useState<'cloud' | 'local'>('local');
  const [localLogs, setLocalLogs] = useState<AuditLogEntry[]>([]);
  const [cloudLogs, setCloudLogs] = useState<AuditLogEntry[]>([]);
  const [isLoadingCloud, setIsLoadingCloud] = useState<boolean>(false);
  const [expandedLogId, setExpandedLogId] = useState<string | null>(null);
  const [copied, setCopied] = useState<boolean>(false);
  const [statusFilter, setStatusFilter] = useState<string>('all');

  const loadLocal = () => setLocalLogs(getLocalAuditLogs());

  const loadCloud = async () => {
    if (!fetchCloudLogs) return;
    setIsLoadingCloud(true);
    try {
      const data = await fetchCloudLogs();
      setCloudLogs(data);
    } catch {
      // ignore
    } finally {
      setIsLoadingCloud(false);
    }
  };

  useEffect(() => {
    if (isOpen) {
      loadLocal();
      if (fetchCloudLogs) loadCloud();
    }
  }, [isOpen]);

  const displayedLogs = useMemo(() => {
    const list = activeTab === 'cloud' ? cloudLogs : localLogs;
    if (statusFilter === 'all') return list;
    return list.filter((l) => l.status === statusFilter);
  }, [activeTab, cloudLogs, localLogs, statusFilter]);

  if (!isOpen) return null;

  const handleExportText = async () => {
    const content = formatAuditLogsAsText(displayedLogs);
    const fileName = `Nhat_Ky_Kiem_Toan_${Date.now()}.txt`;

    if (Capacitor.isNativePlatform()) {
      try {
        const res = await Filesystem.writeFile({
          path: fileName,
          data: content,
          directory: Directory.Cache,
          encoding: Encoding.UTF8,
        });
        await Share.share({ title: 'Xuất Nhật Ký', url: res.uri });
      } catch (e) {
        alert('Lỗi xuất tệp: ' + e);
      }
    } else {
      const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = fileName;
      a.click();
      URL.revokeObjectURL(url);
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-4 bg-slate-950/80 backdrop-blur-xs">
      <div className="bg-slate-900 border border-slate-800 rounded-2xl w-full max-w-2xl max-h-[90vh] flex flex-col text-slate-100 shadow-2xl overflow-hidden animate-in zoom-in-95">
        
        {/* Header */}
        <div className="p-4 border-b border-slate-800 flex items-center justify-between">
          <div className="flex items-center space-x-2">
            <FileText className="w-5 h-5 text-teal-400" />
            <h2 className="font-bold text-sm sm:text-base text-white">Nhật Ký Kiểm Toán & Sự Cố</h2>
          </div>
          <button onClick={onClose} className="p-1 rounded-lg hover:bg-slate-800 text-slate-400 hover:text-white">
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Tab & Filter bar */}
        <div className="px-4 py-2 bg-slate-950/50 border-b border-slate-800 flex flex-wrap items-center justify-between gap-2 text-xs">
          <div className="flex space-x-1 bg-slate-800 p-0.5 rounded-lg">
            {fetchCloudLogs && (
              <button
                onClick={() => setActiveTab('cloud')}
                className={`px-3 py-1 rounded-md font-medium transition-colors ${
                  activeTab === 'cloud' ? 'bg-emerald-600 text-white font-bold' : 'text-slate-400 hover:text-white'
                }`}
              >
                Đám mây ({cloudLogs.length})
              </button>
            )}
            <button
              onClick={() => setActiveTab('local')}
              className={`px-3 py-1 rounded-md font-medium transition-colors ${
                activeTab === 'local' ? 'bg-emerald-600 text-white font-bold' : 'text-slate-400 hover:text-white'
              }`}
            >
              Máy này ({localLogs.length})
            </button>
          </div>

          <div className="flex items-center space-x-2">
            <select
              value={statusFilter}
              onChange={(e) => setStatusFilter(e.target.value)}
              className="bg-slate-800 text-slate-200 border border-slate-700 rounded-lg px-2 py-1 text-xs"
            >
              <option value="all">Tất cả trạng thái</option>
              <option value="success">Thành công</option>
              <option value="error">Lỗi (Error)</option>
              <option value="warning">Cảnh báo</option>
            </select>
            <button
              onClick={handleExportText}
              className="p-1.5 bg-slate-800 hover:bg-slate-700 text-slate-300 rounded-lg"
              title="Xuất file .txt"
            >
              <Download className="w-4 h-4" />
            </button>
          </div>
        </div>

        {/* Logs List */}
        <div className="flex-1 overflow-y-auto p-4 space-y-2.5">
          {displayedLogs.length === 0 ? (
            <div className="text-center py-12 text-slate-500 text-xs">Chưa có bản ghi nhật ký nào.</div>
          ) : (
            displayedLogs.map((log) => {
              const isExpanded = expandedLogId === log.id;
              return (
                <div
                  key={log.id}
                  className={`p-3 rounded-xl border transition-all text-xs ${
                    log.status === 'error'
                      ? 'bg-rose-950/20 border-rose-800/40 text-rose-200'
                      : log.status === 'warning'
                      ? 'bg-amber-950/20 border-amber-800/40 text-amber-200'
                      : 'bg-slate-950/60 border-slate-800 text-slate-300'
                  }`}
                >
                  <div
                    onClick={() => setExpandedLogId(isExpanded ? null : log.id)}
                    className="flex items-start justify-between cursor-pointer"
                  >
                    <div className="flex items-start space-x-2">
                      {log.status === 'error' ? (
                        <AlertCircle className="w-4 h-4 text-rose-400 shrink-0 mt-0.5" />
                      ) : log.status === 'warning' ? (
                        <AlertTriangle className="w-4 h-4 text-amber-400 shrink-0 mt-0.5" />
                      ) : (
                        <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0 mt-0.5" />
                      )}
                      <div>
                        <div className="font-bold text-white text-xs">{log.title}</div>
                        <div className="text-[11px] text-slate-400 mt-0.5">{log.timeStr} • {log.type}</div>
                      </div>
                    </div>
                    {isExpanded ? <ChevronUp className="w-4 h-4 text-slate-500" /> : <ChevronDown className="w-4 h-4 text-slate-500" />}
                  </div>

                  {isExpanded && (
                    <div className="mt-3 pt-3 border-t border-slate-800/80 space-y-2">
                      <p className="text-xs text-slate-300 leading-relaxed">{log.summary}</p>
                      {log.errorMessage && (
                        <div className="p-2 bg-rose-950/50 rounded-lg text-rose-300 font-mono text-[11px]">
                          {log.errorMessage}
                        </div>
                      )}
                      {log.details && (
                        <pre className="p-2 bg-slate-950 rounded-lg text-[10px] font-mono text-emerald-400 overflow-x-auto">
                          {JSON.stringify(log.details, null, 2)}
                        </pre>
                      )}
                    </div>
                  )}
                </div>
              );
            })
          )}
        </div>
      </div>
    </div>
  );
};
```

```typescript
import { recordAuditLog } from './utils/auditLogger';

try {
  // Thực hiện hành động...

    type: 'LOGIN_SUCCESS',
    title: 'Đăng nhập thành công',
    status: 'success',
    summary: 'Người dùng tuanta@gmail.com đã đăng nhập vào hệ thống',

  recordAuditLog({
    type: 'NETWORK_ERROR',
```
